### 2.2 Skill 系统

```python
# skills/code-review.md
# ---
# name: code-review
# description: Review code for bugs, style, and security issues
# triggers:
#   - "review this code"
#   - "检查代码"
# ---
# ## Instructions
# 1. Check for security vulnerabilities (SQL injection, XSS)
# 2. Review code style and naming conventions
# 3. Suggest performance improvements
# 4. Output in markdown with file references

class SkillLoader:
    """扫描 skills/ 目录，解析 .md 元数据"""

    def __init__(self, skills_dir="skills/"):
        self.skills = {}  # name → {meta, content}

    def load_all(self):
        for f in os.listdir(self.skills_dir):
            if f.endswith(".md"):
                meta, content = self._parse(f)
                self.skills[meta["name"]] = {
                    "description": meta.get("description", ""),
                    "triggers": meta.get("triggers", []),
                    "instructions": content,
                }

    def _parse(self, filename):
        # 解析 YAML frontmatter
        with open(filename) as f:
            text = f.read()
        if text.startswith("---"):
            _, front, body = text.split("---", 2)
            meta = yaml.safe_load(front)
            return meta, body.strip()
        return {}, text

    def match(self, user_input: str) -> list[str]:
        """根据用户输入匹配应加载的 skills"""
        matched = []
        for name, skill in self.skills.items():
            if any(t in user_input for t in skill["triggers"]):
                matched.append(name)
        return matched

    def inject_prompt(self, skill_names: list[str], pb: PromptBuilder):
        """将匹配的 skill 注入 PromptBuilder context 层"""
        for name in skill_names:
            skill = self.skills[name]
            pb.load_context(
                f"[Skill: {name}] {skill['description']}\n{skill['instructions']}"
            )
```

**使用流程：**

```python
loader = SkillLoader()
loader.load_all()

# 用户说 "review this code" → 自动匹配 code-review skill
matched = loader.match("please review this code for bugs")
# → ["code-review"]

pb = PromptBuilder()
pb.set_personality("你是通用助手")
loader.inject_prompt(matched, pb)
# → system prompt 里自动包含 code-review 的 instructions
```

### 2.3 Slash Command 系统

```python
# Hermes Agent 命令注册模式
COMMANDS = {}

def register_command(name: str, description: str):
    """装饰器：注册斜杠命令"""
    def decorator(fn):
        COMMANDS[name] = {"func": fn, "desc": description}
        return fn
    return decorator

@register_command("memory", "读写持久化记忆")
def cmd_memory(args, agent):
    """用法: /memory save key value 或 /memory load key"""
    parts = args.split()
    if parts[0] == "save":
        agent.memory.save(parts[1], " ".join(parts[2:]))
        return f"已保存: {parts[1]}"
    elif parts[0] == "load":
        return agent.memory.load(parts[1])
    elif parts[0] == "list":
        return agent.memory.list_keys()

@register_command("model", "切换 LLM 模型")
def cmd_model(args, agent):
    """用法: /model deepseek-chat 或 /model gpt-4o"""
    agent.set_provider(args.strip())
    return f"已切换到 {args.strip()}"

@register_command("skill", "加载/列出 Skill")
def cmd_skill(args, agent):
    """用法: /skill list 或 /skill load code-review"""
    if args.strip() == "list":
        return "\n".join(agent.skills.list())
    agent.skills.load(args.strip())
    return f"已加载 skill: {args}"

@register_command("file", "上传附件到当前会话")
def cmd_file(args, agent):
    """用法: /file path/to/document.pdf"""
    # MinerU 转 MD → 注入当前会话上下文
    result = mineru.convert(args)
    agent.context.add(result.markdown)
    return f"已加载文件: {args} ({len(result.markdown)} 字)"

# 路由
def handle_message(user_input, agent):
    if user_input.startswith("/"):
        parts = user_input.split(" ", 1)
        cmd = parts[0][1:]
        args = parts[1] if len(parts) > 1 else ""
        if cmd in COMMANDS:
            return COMMANDS[cmd]["func"](args, agent)
    return agent.chat(user_input)
```
